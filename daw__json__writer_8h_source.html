54 template<typename WritableType, typename StackType, auto... PolicyFlags>

60 StackType m_stack{ };
61 json_writer_details::json_writer_states m_current_state =
62 json_writer_details::json_writer_states::json_writer_nothing;
63 bool m_is_first = true;
64 bool m_is_key_written = false;

67 constexpr void
68 push_state( json_writer_details::json_writer_states new_state ) {
69 m_stack.push_back( m_current_state );

71 }

73 constexpr void pop_state( ) {
74 daw_json_ensure( not m_stack.empty( ), ErrorReason::OutputError );
75 m_current_state = m_stack.back( );
76 m_stack.pop_back( );

79 constexpr void write_item_prefix( ) {
80 daw_json_ensure( not m_is_key_written, ErrorReason::OutputError );
81 if( not m_is_first ) {
82 m_writer.put( ',' );
83 } else {
84 m_is_first = false;
85 }
86 m_writer.next_member( );
87 }

89 constexpr void prepare_value( ) {
90 if( m_current_state ==
91 json_writer_details::json_writer_states::json_writer_nothing ) {
92 daw_json_ensure( m_is_first, ErrorReason::OutputError );
93 return;
94 }
95 if( m_current_state ==
96 json_writer_details::json_writer_states::json_writer_object ) {
97 daw_json_ensure( m_is_key_written, ErrorReason::OutputError );
98 m_is_key_written = false;
99 return;
100 }
101 if( m_current_state !=
102 json_writer_details::json_writer_states::json_writer_nothing ) {
103 write_item_prefix( );
104 }
105 }

124 constexpr void finalize( ) {
125 if( m_is_key_written ) {
126 write_value( nullptr );
127 }
128 while( m_current_state !=
129 json_writer_details::json_writer_states::json_writer_nothing ) {
130 switch( m_current_state ) {
131 case json_writer_details::json_writer_states::json_writer_object:
132 close_object( );
133 break;
134 case json_writer_details::json_writer_states::json_writer_array:
135 close_array( );
136 break;
137 case json_writer_details::json_writer_states::json_writer_nothing:
138 default:
139 break;
140 }
141 }
142 }

144 constexpr void reset( ) {
145 finalize( );
146 m_current_state =
147 json_writer_details::json_writer_states::json_writer_nothing;
148 m_is_first = true;
149 m_is_key_written = false;
150 }

157 constexpr void open_object( ) {
158 prepare_value( );
159 push_state(
160 json_writer_details::json_writer_states::json_writer_object );
161 m_writer.put( '{' );
162 m_writer.add_indent( );

166 constexpr void close_object( ) {
167 daw_json_ensure(
168 m_current_state ==
169 json_writer_details::json_writer_states::json_writer_object,
170 ErrorReason::OutputError );
171 if( m_is_key_written ) {
172 write_value( nullptr );

377 template<typename JsonClass = use_default, typename T>
378 constexpr void write_number( T const &value ) {
379 using JsonMember =
380 json_writer_details::json_write_value_class_t<JsonClass, T>;
381 constexpr JsonBaseParseTypes json_base_type =
382 JsonMember::underlying_json_type;
383 static_assert(
384 json_base_type == JsonBaseParseTypes::Number or
385 json_base_type == JsonBaseParseTypes::Bool,
386 "The underlying mapping must be a number or boolean type" );
387 if constexpr( json_base_type == JsonBaseParseTypes::Bool ) {
388 write_value<JsonClass>( static_cast<bool>( value ) ? 1 : 0 );
389 return;
390 } else {
391 write_value<JsonClass>( value );
392 }
393 }

396 template<typename JsonClass = use_default, typename T>
397 constexpr void write_string( T const &value ) {
398 using JsonMember =
399 json_writer_details::json_write_value_class_t<JsonClass, T>;
400 constexpr JsonBaseParseTypes json_base_type =
401 JsonMember::underlying_json_type;
402 if constexpr( json_base_type == JsonBaseParseTypes::String ) {
403 write_value<JsonClass>( value );
404 } else if constexpr( json_base_type == JsonBaseParseTypes::Bool or
405 json_base_type == JsonBaseParseTypes::Number ) {
406 prepare_value( );
407 m_writer.put( '"' );
408 to_json<JsonClass>( value, m_writer.get( ) );
409 m_writer.put( '"' );
410 m_is_first = false;
411 } else {
412 auto const tmp = to_json( value );
413 do_write_value<json_string_no_name<>>( tmp );
414 }
415 }
