56 StackType m_stack{ };
57 json_writer_details::json_writer_states m_current_state =

59 bool m_is_first = true;
60 bool m_is_key_written = false;

63 constexpr void
64 push_state( json_writer_details::json_writer_states new_state ) {
65 m_stack.push_back( m_current_state );
66 m_current_state = new_state;
67 }

69 constexpr void pop_state( ) {
70 daw_json_ensure( not m_stack.empty( ), ErrorReason::OutputError );
71 m_current_state = m_stack.back( );
72 m_stack.pop_back( );

75 constexpr void write_item_prefix( ) {
76 daw_json_ensure( not m_is_key_written, ErrorReason::OutputError );
77 if( not m_is_first ) {
78 m_writer.put( ',' );
79 } else {
80 m_is_first = false;
81 }
82 m_writer.next_member( );
83 }

85 constexpr void prepare_value( ) {
86 if( m_current_state ==
87 json_writer_details::json_writer_states::json_writer_nothing ) {
88 daw_json_ensure( m_is_first, ErrorReason::OutputError );
89 return;
90 }
91 if( m_current_state ==
92 json_writer_details::json_writer_states::json_writer_object ) {
93 daw_json_ensure( m_is_key_written, ErrorReason::OutputError );
94 m_is_key_written = false;
95 return;
96 }
97 if( m_current_state !=
98 json_writer_details::json_writer_states::json_writer_nothing ) {
99 write_item_prefix( );
100 }
101 }

103 template<typename T>
104 constexpr void do_write_value( T const &value ) {
105 prepare_value( );
106 to_json( value, m_writer.get( ) );
107 m_is_first = false;

120 constexpr void finalize( ) {
121 if( m_is_key_written ) {
122 write_value( nullptr );
123 }
124 while( m_current_state !=
125 json_writer_details::json_writer_states::json_writer_nothing ) {
126 switch( m_current_state ) {
127 case json_writer_details::json_writer_states::json_writer_object:
128 close_object( );
129 break;
130 case json_writer_details::json_writer_states::json_writer_array:
131 close_array( );
132 break;
133 case json_writer_details::json_writer_states::json_writer_nothing:
134 default:
135 break;
136 }
137 }
138 }
139
140 constexpr void reset( ) {
141 finalize( );
142 m_current_state =
143 json_writer_details::json_writer_states::json_writer_nothing;
144 m_is_first = true;
145 m_is_key_written = false;
146 }

153 constexpr void open_object( ) {
154 prepare_value( );
155 push_state(
156 json_writer_details::json_writer_states::json_writer_object );
157 m_writer.put( '{' );
158 m_writer.add_indent( );
159 m_is_first = true;
160 }
161
162 constexpr void close_object( ) {
163 daw_json_ensure(
164 m_current_state ==
165 json_writer_details::json_writer_states::json_writer_object,
166 ErrorReason::OutputError );
167 if( m_is_key_written ) {
168 write_value( nullptr );
169 }
170 m_writer.del_indent( );
171 if( not m_is_first ) {
172 m_writer.next_member( );
173 }
174 m_writer.put( '}' );
175 pop_state( );
176 m_is_first = false;
177 }

185 m_writer.put( '[' );
186 }

189 daw_json_ensure(
190 m_current_state ==

192 ErrorReason::OutputError );

194 if( not m_is_first ) {
195 m_writer.next_member( );
196 }
197 m_writer.put( ']' );
198 pop_state( );
199 m_is_first = false;

206 template<typename T, typename... Ts>
207 constexpr void write_value( T const &value, Ts const &...values ) {
208 daw_json_ensure(
209 m_current_state !=
210 json_writer_details::json_writer_states::json_writer_object or
211 m_is_key_written,
212 ErrorReason::OutputError );
213 daw_json_ensure(
214 m_current_state !=
215 json_writer_details::json_writer_states::json_writer_nothing or
216 m_is_first,
217 ErrorReason::OutputError );
218 if constexpr( sizeof...( Ts ) == 0 ) {
219 do_write_value( value );
220 } else {
221 open_array( );
222 write_array_values( value, values... );
223 close_array( );
224 }
225 }

358 template<typename JsonClass = use_default, typename T>
359 constexpr void write_number( T const &value ) {
360 using JsonMember =
361 json_writer_details::json_write_value_class_t<JsonClass, T>;
362 constexpr JsonBaseParseTypes json_base_type =
363 JsonMember::underlying_json_type;
364 static_assert(
365 json_base_type == JsonBaseParseTypes::Number or
366 json_base_type == JsonBaseParseTypes::Bool,
367 "The underlying mapping must be a number or boolean type" );
368 if constexpr( json_base_type == JsonBaseParseTypes::Bool ) {
369 write_value( static_cast<bool>( value ) ? 1 : 0 );
370 return;
371 } else {
372 write_value( value );
373 }
374 }

376 template<typename JsonClass = use_default, typename T>
377 constexpr void write_string( T const &value ) {
378 using JsonMember =
379 json_writer_details::json_write_value_class_t<JsonClass, T>;
380 constexpr JsonBaseParseTypes json_base_type =
381 JsonMember::underlying_json_type;
382 static_assert(
383 json_base_type == JsonBaseParseTypes::Number or
384 json_base_type == JsonBaseParseTypes::Bool or
385 json_base_type == JsonBaseParseTypes::String,
386 "The underlying mapping must be a number, boolean, or string type" );
387 if constexpr( json_base_type == JsonBaseParseTypes::String ) {
388 write_value( value );
389 return;
390 } else {
391 prepare_value( );
392 m_writer.put( '"' );
393 to_json( value, m_writer.get( ) );
394 m_writer.put( '"' );
395 m_is_first = false;
396 }
