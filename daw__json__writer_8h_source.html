39 template<typename WriterType, typename StackType, auto... PolicyFlags>
40 class json_writer_t {
41 using iterator_t =
42 decltype( json_details::make_output_iterator<PolicyFlags...>(
43 std::declval<WriterType>( ) ) );
44
45 /*std::vector<json_writer_states>*/ StackType m_stack{ };
46 json_writer_states m_current_state =
47 json_writer_states::json_writer_nothing;
48 bool m_is_first = true;
49 bool m_is_key_written = false;
50 iterator_t m_writer;
51
52 constexpr void push_state( json_writer_states new_state ) {
53 m_stack.push_back( m_current_state );
54 m_current_state = new_state;
55 }
56
57 constexpr void pop_state( ) {
58 daw_json_ensure( not m_stack.empty( ), ErrorReason::OutputError );
59 m_current_state = m_stack.back( );
60 m_stack.pop_back( );
61 }
62
63 constexpr void write_item_prefix( ) {
64 daw_json_ensure( not m_is_key_written, ErrorReason::OutputError );
65 if( not m_is_first ) {
66 m_writer.put( ',' );
67 } else {
68 m_is_first = false;
69 }
70 m_writer.next_member( );
71 }
72
73 constexpr void prepare_value( ) {
74 if( m_current_state == json_writer_states::json_writer_nothing ) {
75 daw_json_ensure( m_is_first, ErrorReason::OutputError );
76 return;
77 }
78 if( m_current_state == json_writer_states::json_writer_object ) {
79 daw_json_ensure( m_is_key_written, ErrorReason::OutputError );
80 m_is_key_written = false;
81 return;
82 }
83 if( m_current_state != json_writer_states::json_writer_nothing ) {
84 write_item_prefix( );
85 }
86 }
87
88 template<typename T>
89 constexpr void do_write_value( T const &value ) {
90 prepare_value( );
91 to_json( value, m_writer.get( ) );
92 m_is_first = false;
93 }

104 DAW_CPP20_CX_DTOR ~json_writer_t( ) noexcept(
105 noexcept( m_writer.put( ' ' ) ) ) {
106 if( m_is_key_written ) {
107 write_value( nullptr );
108 }
109 while( m_current_state != json_writer_states::json_writer_nothing ) {
110 switch( m_current_state ) {
111 case json_writer_states::json_writer_object:
112 close_object( );
113 break;
114 case json_writer_states::json_writer_array:
115 close_array( );
116 break;
117 case json_writer_states::json_writer_nothing:
118 default:
119 break;
120 }
121 }
122 }

124 constexpr void open_object( ) {
125 prepare_value( );
126 push_state( json_writer_states::json_writer_object );
127 m_writer.put( '{' );
128 m_writer.add_indent( );
129 m_is_first = true;
130 }
131
132 constexpr void close_object( ) {
133 daw_json_ensure( m_current_state ==
134 json_writer_states::json_writer_object,
135 ErrorReason::OutputError );
136 if( m_is_key_written ) {
137 write_value( nullptr );
138 }
139 m_writer.del_indent( );
140 if( not m_is_first ) {
141 m_writer.next_member( );
142 }
143 m_writer.put( '}' );
144 pop_state( );
145 m_is_first = false;
146 }
147
148 constexpr void open_array( ) {
149 prepare_value( );
150 push_state( json_writer_states::json_writer_array );
151 m_is_first = true;
152 m_writer.add_indent( );
153 m_writer.put( '[' );
154 }
155
156 constexpr void close_array( ) {
157 daw_json_ensure( m_current_state ==
158 json_writer_states::json_writer_array,
159 ErrorReason::OutputError );
160 m_writer.del_indent( );
161 if( not m_is_first ) {
162 m_writer.next_member( );
163 }
164 m_writer.put( ']' );
165 pop_state( );
166 m_is_first = false;
167 }

169 // Write a value similar to to_json.
170 template<typename T>
171 constexpr void write_value( T const &value ) {
172 daw_json_ensure( m_current_state !=
173 json_writer_states::json_writer_object or
174 m_is_key_written,
175 ErrorReason::OutputError );
176 daw_json_ensure( m_current_state !=
177 json_writer_states::json_writer_nothing or
178 m_is_first,
179 ErrorReason::OutputError );
180 do_write_value( value );
181 }

193 do_write_value( daw::string_view( str, N - 1 ) );
194 }

196 constexpr void write_value( std::nullptr_t ) {
197 daw_json_ensure( m_current_state !=
198 json_writer_states::json_writer_nothing or
199 m_is_first,
200 ErrorReason::OutputError );
201 prepare_value( );
202 m_writer.write( "null" );
203 m_is_first = false;
204 }

206 // Must be in array state
207 template<
208 typename Range DAW_ENABLEIF( daw::traits::is_container_like_v<Range> )>

210 constexpr void write_values( Range const &values ) {
211 daw_json_ensure( m_current_state ==

237 constexpr void add_key( daw::string_view name ) {
238 if( m_is_key_written ) {
239 write_value( nullptr );
240 }
241 daw_json_ensure( m_current_state ==
242 json_writer_states::json_writer_object,
243 ErrorReason::OutputError );
244
245 write_item_prefix( );
246 m_writer.write( "\"", name, "\":", m_writer.space );
247 m_is_key_written = true;
248 }
