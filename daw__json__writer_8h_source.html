46 template<typename WriterType, typename StackType, auto... PolicyFlags>
47 class json_writer_t {
48 using iterator_t =
49 decltype( json_details::make_output_iterator<PolicyFlags...>(
50 std::declval<WriterType>( ) ) );
51
52 /*std::vector<json_writer_states>*/ StackType m_stack{ };
53 json_writer_states m_current_state =
54 json_writer_states::json_writer_nothing;
55 bool m_is_first = true;
56 bool m_is_key_written = false;
57 iterator_t m_writer;
58
59 constexpr void push_state( json_writer_states new_state ) {
60 m_stack.push_back( m_current_state );
61 m_current_state = new_state;
62 }
63
64 constexpr void pop_state( ) {
65 daw_json_ensure( not m_stack.empty( ), ErrorReason::OutputError );
66 m_current_state = m_stack.back( );
67 m_stack.pop_back( );
68 }
69
70 constexpr void write_item_prefix( ) {
71 daw_json_ensure( not m_is_key_written, ErrorReason::OutputError );
72 if( not m_is_first ) {
73 m_writer.put( ',' );
74 } else {
75 m_is_first = false;
76 }
77 m_writer.next_member( );
78 }
79
80 constexpr void prepare_value( ) {
81 if( m_current_state == json_writer_states::json_writer_nothing ) {
82 daw_json_ensure( m_is_first, ErrorReason::OutputError );
83 return;
84 }
85 if( m_current_state == json_writer_states::json_writer_object ) {
86 daw_json_ensure( m_is_key_written, ErrorReason::OutputError );
87 m_is_key_written = false;
88 return;
89 }
90 if( m_current_state != json_writer_states::json_writer_nothing ) {
91 write_item_prefix( );
92 }
93 }

95 template<typename T>
96 constexpr void do_write_value( T const &value ) {
97 prepare_value( );
98 to_json( value, m_writer.get( ) );
99 m_is_first = false;
100 }

103 explicit constexpr json_writer_t( WriterType &writer )
104 : m_writer(

111 constexpr void finalize( ) {
112 if( m_is_key_written ) {
113 write_value( nullptr );
114 }
115 while( m_current_state != json_writer_states::json_writer_nothing ) {
116 switch( m_current_state ) {
117 case json_writer_states::json_writer_object:
118 close_object( );
119 break;
120 case json_writer_states::json_writer_array:
121 close_array( );
122 break;
123 case json_writer_states::json_writer_nothing:
124 default:
125 break;
126 }
127 }
128 }
129
130 constexpr void reset( ) {
131 finalize( );
132 m_current_state = json_writer_states::json_writer_nothing;
133 m_is_first = true;
134 m_is_key_written = false;
135 }

142 constexpr void open_object( ) {
143 prepare_value( );
144 push_state( json_writer_states::json_writer_object );
145 m_writer.put( '{' );
146 m_writer.add_indent( );
147 m_is_first = true;
148 }
149
150 constexpr void close_object( ) {
151 daw_json_ensure( m_current_state ==
152 json_writer_states::json_writer_object,
153 ErrorReason::OutputError );
154 if( m_is_key_written ) {
155 write_value( nullptr );
156 }
157 m_writer.del_indent( );
158 if( not m_is_first ) {
159 m_writer.next_member( );
160 }
161 m_writer.put( '}' );
162 pop_state( );
163 m_is_first = false;
164 }
165
166 constexpr void open_array( ) {
167 prepare_value( );
168 push_state( json_writer_states::json_writer_array );
169 m_is_first = true;
170 m_writer.add_indent( );
171 m_writer.put( '[' );
172 }
173
174 constexpr void close_array( ) {
175 daw_json_ensure( m_current_state ==
176 json_writer_states::json_writer_array,
177 ErrorReason::OutputError );
178 m_writer.del_indent( );
179 if( not m_is_first ) {
180 m_writer.next_member( );
181 }
182 m_writer.put( ']' );
183 pop_state( );
184 m_is_first = false;
185 }

188 template<typename T>
189 constexpr void write_value( T const &value ) {
190 daw_json_ensure( m_current_state !=
191 json_writer_states::json_writer_object or
192 m_is_key_written,
193 ErrorReason::OutputError );
194 daw_json_ensure( m_current_state !=
195 json_writer_states::json_writer_nothing or
196 m_is_first,
197 ErrorReason::OutputError );
198 do_write_value( value );
199 }

215 template<typename JsonClass = use_default, typename T>
216 constexpr void write_number( T const &value ) {
217 using JsonMember =
218 json_writer_details::json_write_value_class_t<JsonClass, T>;
219 constexpr JsonBaseParseTypes json_base_type =
220 JsonMember::underlying_json_type;
221 static_assert(
222 json_base_type == JsonBaseParseTypes::Number or
223 json_base_type == JsonBaseParseTypes::Bool,
224 "The underlying mapping must be a number or boolean type" );
225 if constexpr( json_base_type == JsonBaseParseTypes::Bool ) {
226 write_value( static_cast<bool>( value ) ? 1 : 0 );
227 return;
228 }
229 write_value( value );
230 }
231
232 template<typename JsonClass = use_default, typename T>
233 constexpr void write_string( T const &value ) {
234 using JsonMember =
235 json_writer_details::json_write_value_class_t<JsonClass, T>;
236 constexpr JsonBaseParseTypes json_base_type =
237 JsonMember::underlying_json_type;
238 static_assert(
239 json_base_type == JsonBaseParseTypes::Number or
240 json_base_type == JsonBaseParseTypes::Bool or
241 json_base_type == JsonBaseParseTypes::String,
242 "The underlying mapping must be a number, boolean, or string type" );
243 if constexpr( json_base_type == JsonBaseParseTypes::String ) {
244 write_value( value );
245 return;
246 }
247 prepare_value( );
248 m_writer.put( '"' );
249 to_json( value, m_writer.get( ) );
250 m_writer.put( '"' );
251 m_is_first = false;
252 }

307 constexpr void add_key( daw::string_view name ) {
308 if( m_is_key_written ) {
309 write_value( nullptr );
310 }
311 daw_json_ensure( m_current_state ==
312 json_writer_states::json_writer_object,
313 ErrorReason::OutputError );
314
315 write_item_prefix( );
316 m_writer.write( "\"", name, "\":", m_writer.space );
317 m_is_key_written = true;
318 }
