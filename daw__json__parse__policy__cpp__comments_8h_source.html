30 template<typename ParseState>
31 DAW_ATTRIB_FLATINLINE static constexpr void
32 skip_comments_unchecked( ParseState &parse_state ) {
33 while( parse_state.front( ) == '/' ) {
34 switch( *( parse_state.first + 1 ) ) {
35 case '/':
36 parse_state.template move_to_next_of_unchecked<'\n'>( );
37 parse_state.remove_prefix( );
38 break;
39 case '*':
40 parse_state.remove_prefix( 2 );
41 while( true ) {
42 parse_state.template move_to_next_of_unchecked<'*'>( );
43 parse_state.remove_prefix( );
44 if( parse_state.front( ) == '/' ) {
45 parse_state.remove_prefix( );
46 break;
47 }
48 parse_state.remove_prefix( );
49 }
50 break;
51 default:
52 return;
53 }
54 }
55 }

57 template<typename ParseState>
58 DAW_ATTRIB_FLATINLINE static constexpr void
59 skip_comments_checked( ParseState &parse_state ) {
60 while( parse_state.has_more( ) and parse_state.front( ) == '/' ) {
61 if( not parse_state.has_more( ) ) {
62 return;
63 }
64 switch( *( parse_state.first + 1 ) ) {
65 case '/':
66 parse_state.template move_to_next_of_checked<'\n'>( );
67 if( parse_state.has_more( ) ) {
68 parse_state.remove_prefix( );
69 }
70 break;
71 case '*':
72 parse_state.remove_prefix( 2 );
73 while( parse_state.has_more( ) ) {
74 parse_state.template move_to_next_of_checked<'*'>( );
75 if( parse_state.has_more( ) ) {
76 parse_state.remove_prefix( );
77 }
78 if( not parse_state.has_more( ) ) {
79 break;
80 } else if( parse_state.front( ) == '/' ) {
81 parse_state.remove_prefix( );
82 break;
83 }
84 parse_state.remove_prefix( );
85 }
86 break;
87 default:
88 return;
89 }
90 }

130 DAW_ATTRIB_FLATINLINE static constexpr void
131 move_to_next_of( ParseState &parse_state ) {
132 skip_comments( parse_state );
133 daw_json_assert_weak( parse_state.has_more( ),
134 ErrorReason::UnexpectedEndOfData,
135 parse_state );

149 template<json_details::SkipBracketedType BracketedType,
150 typename ParseState>
151 DAW_ATTRIB_FLATINLINE static constexpr ParseState
152 skip_bracketed_item_checked( ParseState &parse_state ) {
153 constexpr char PrimLeft =
154 BracketedType == json_details::SkipBracketedType::Class ? '{' : '[';
155 using PrimRight = daw::constant<PrimLeft == '{' ? '}' : ']'>;
156 using SecLeft = daw::constant<PrimLeft == '{' ? '[' : '{'>;
157 using SecRight = daw::constant<SecLeft::value == '{' ? '}' : ']'>;
158
159 // Not checking for Left as it is required to be skipped already
160 auto result = parse_state;
161 std::size_t cnt = 0;
162 std::uint32_t prime_bracket_count = 1;
163 std::uint32_t second_bracket_count = 0;
164 auto ptr_first = daw::not_null<char const *>( parse_state.first );
165 auto const ptr_last = daw::not_null<char const *>( parse_state.last );
166 if( DAW_UNLIKELY( ptr_first >= ptr_last ) ) {
167 return result;
168 }
169 if( *ptr_first == PrimLeft ) {
170 ++ptr_first;
171 }
172 while( DAW_LIKELY( ptr_first < ptr_last ) ) {
173 switch( *ptr_first ) {
174 case '\\':
175 ++ptr_first;
176 break;
177 case '"':
178 ++ptr_first;
179 ptr_first = json_details::mem_skip_until_end_of_string<
180 ParseState::is_unchecked_input,
181 typename ParseState::exec_tag_t>( ptr_first, ptr_last );
182 daw_json_ensure( ptr_first < ptr_last,
183 ErrorReason::UnexpectedEndOfData,
184 parse_state );
185 break;
186 case ',':
187 if( prime_bracket_count == 1 and second_bracket_count == 0 ) {
188 ++cnt;
189 }
190 break;
191 case PrimLeft:
192 ++prime_bracket_count;
193 break;
194 case PrimRight::value:
195 --prime_bracket_count;
196 if( prime_bracket_count == 0 ) {
197 daw_json_ensure( second_bracket_count == 0,
198 ErrorReason::InvalidBracketing,
199 parse_state );
200 ++ptr_first;
201 // We include the close primary bracket in the range so that
202 // subsequent parsers have a terminator inside their range
203 result.last = ptr_first;
204 result.counter = cnt;
205 parse_state.first = ptr_first;
206 return result;
207 }
208 break;
209 case SecLeft::value:
210 ++second_bracket_count;
211 break;
212 case SecRight::value:
213 --second_bracket_count;
214 break;
215 case '/':
216 ++ptr_first;
217 daw_json_ensure( ptr_first < ptr_last,
218 ErrorReason::UnexpectedEndOfData,
219 parse_state );
220 switch( *ptr_first ) {
221 case '/':
222 ++ptr_first;
223 while( ( ptr_last - ptr_first ) > 1 and *ptr_first != '\n' ) {
224 ++ptr_first;
225 }
226 break;
227 case '*':
228 ++ptr_first;
229 while( ( ptr_last - ptr_first ) >= 3 and *ptr_first != '*' and
230 *std::next( ptr_first.get( ) ) != '/' ) {
231 ++ptr_first;
232 }
233 break;
234 default:
235 ++ptr_first;
236 }
237 break;
238 }
239 ++ptr_first;
240 }
241 daw_json_assert_weak( ( prime_bracket_count == 0 ) &
242 ( second_bracket_count == 0 ),
243 ErrorReason::InvalidBracketing,
244 parse_state );
245 // We include the close primary bracket in the range so that subsequent
246 // parsers have a terminator inside their range
247 result.last = ptr_first;
248 result.counter = cnt;
249 parse_state.first = ptr_first;
250 return result;
251 }

288 ++cnt;
