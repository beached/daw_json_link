109 template<typename T>
110 concept TypeAnnotatable =
111 std::is_class_v<T> or std::is_enum_v<T> or std::is_union_v<T>;

115 template<typename Annotation, typename T>
116 consteval std::optional<Annotation> get_type_annotation( ) {
117 if constexpr( TypeAnnotatable<T> ) {
118 for( auto annot : annotations_of( ^^T ) ) {
119 if( remove_cv( type_of( annot ) ) == ^^Annotation ) {
120 return extract<Annotation>( annot );
121 }
122 }
123 }
124 return std::nullopt;
125 }

127 template<typename AnnotationBase, typename T>
128 consteval bool has_type_annotation_with_base( ) {
129 if constexpr( TypeAnnotatable<T> ) {
130 return not annotations_of_with_base_type( ^^T, ^^AnnotationBase )
131 .empty( );
132 } else {
133 return false;

169 template<HasTypeLevelAnnotation T>
170 consteval auto type_annotation_mapping( ) {
171 static_assert( not( std::is_enum_v<T> and
172 has_type_annotation_with_base<reflect_base_t, T>( ) ),
173 "reflect is only valid on class types. Use "
174 "reflect.enum_string to map an enum as a string" );
175 static_assert( not get_type_annotation<refl_rename, T>( ),
176 "reflect.rename is only valid on members, not types" );
177 static_assert( not has_type_annotation_with_base<refl_ignored_base, T>( ),
178 "reflect.ignored is only valid on members, not types" );
179 static_assert( std::is_enum_v<T> or
180 not get_type_annotation<refl_enum_string, T>( ),
181 "reflect.enum_string is only valid on enum types or "
182 "members of enum type" );
183 static_assert( std::is_enum_v<T> or
184 not get_type_annotation<refl_enum_string_or_number, T>( ),
185 "reflect.enum_string_or_number is only valid on enum "
186 "types or members of enum type" );
187 static_assert( not( EnumStringAnnotated<T> and MapAsAnnotated<T> ),
188 "Do not use reflect.enum_string and reflect.map_as "
189 "at the same time" );
190 static_assert( not( EnumStringOrNumberAnnotated<T> and MapAsAnnotated<T> ),
191 "Do not use reflect.enum_string_or_number and "
192 "reflect.map_as at the same time" );
193 static_assert(
194 not( EnumStringOrNumberAnnotated<T> and EnumStringAnnotated<T> ),
195 "Do not use reflect.enum_string_or_number and reflect.enum_string at "
196 "the same time" );
197 if constexpr( EnumStringAnnotated<T> ) {
198 static constexpr auto annot = get_type_annotation<refl_enum_string, T>( );
199 return daw::traits::identity<enum_string<T, annot->Options>>{ };
200 } else if constexpr( EnumStringOrNumberAnnotated<T> ) {
201 static constexpr auto annot =
202 get_type_annotation<refl_enum_string_or_number, T>( );
203 return daw::traits::identity<
204 enum_string_or_number<T, annot->NumberOptions>>{ };
205 } else if constexpr( MapAsAnnotated<T> ) {
206 static constexpr auto annot = get_type_annotation<refl_map_as, T>( );
207 using mapping_t = typename[:annot->type:];
208 static_assert( json_details::is_a_json_type_v<mapping_t>,
209 "A type level reflect.map_as<JsonType> requires a JSON "
210 "type, e.g. json_number_no_name<int>" );
211 static_assert( json_details::is_no_name_v<mapping_t>,
212 "A type level reflect.map_as<JsonType> requires a no "
213 "name JSON type, e.g. json_number_no_name<int>. The "
214 "member name is supplied where the type is used" );
215 return daw::traits::identity<mapping_t>{ };
216 } else {
217 // Unreachable, the checks above will have failed
218 return daw::traits::identity<void>{ };
219 }
220 }

224 typename decltype( type_annotation_mapping<T>( ) )::type;
