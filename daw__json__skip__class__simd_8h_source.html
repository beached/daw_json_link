87 template<SkipBracketedType BracketedType, typename ParseState>
88 [[nodiscard]] DAW_ATTRIB_FLATTEN DAW_JSON_SIMD_CONSTEXPR ParseState
89 skip_bracketed_item_simd( ParseState &parse_state ) {
90 constexpr char primary_left =
91 BracketedType == SkipBracketedType::Class ? '{' : '[';
92 constexpr char primary_right =
93 BracketedType == SkipBracketedType::Class ? '}' : ']';
94 constexpr char secondary_left =
95 BracketedType == SkipBracketedType::Class ? '[' : '{';
96 constexpr char secondary_right =
97 BracketedType == SkipBracketedType::Class ? ']' : '}';
98
99 using namespace skip_bracketed_item_simd_details;
100 auto ptr_first = parse_state.first;
101 auto const ptr_last = parse_state.last;
102 if( DAW_UNLIKELY( ptr_first >= ptr_last ) ) {
103 return parse_state;
104 }
105
106 auto result = parse_state;
107 std::size_t count = 0;
108 std::uint32_t primary_depth = 1;
109 std::uint32_t secondary_depth = 0;
110 bool in_string = false;
111 bool escaped = false;
112
113 if( *ptr_first == primary_left ) {
114 ++ptr_first;
115 }
116
117 while( ptr_first < ptr_last ) {
118 auto const remaining =
119 static_cast<std::size_t>( ptr_last - ptr_first );
120 auto const lane_count =
121 remaining < block_size ? remaining : block_size;
122 auto const input = load( ptr_first, lane_count );
123 auto const valid_bits = low_bits( lane_count );
124 auto const backslash_bits =
125 ( input == splat( '\\' ) ).to_ullong( ) & valid_bits;
126
127 constexpr std::uint64_t odd_bits = 0xAAAAAAAAAAAAAAAAULL;
128 auto const previous_escaped = escaped ? std::uint64_t{ 1 } : 0;
129 auto const potential_escape = backslash_bits & ~previous_escaped;
130 auto const maybe_escaped = potential_escape << 1U;
131 auto const escape_and_terminal =
132 ( ( maybe_escaped | odd_bits ) - potential_escape ) ^ odd_bits;
133 auto const escaped_bits =
134 ( escape_and_terminal ^ ( backslash_bits | previous_escaped ) ) &
135 valid_bits;
136 auto const escape_bits = escape_and_terminal & backslash_bits;
137 escaped = last_bit( escape_bits, lane_count );
138
139 auto const quote_bits =
140 ( input == splat( '"' ) ).to_ullong( ) & ~escaped_bits & valid_bits;
141 auto const in_string_bits =
142 ( prefix_xor( quote_bits ) ^
143 ( in_string ? valid_bits : std::uint64_t{ 0 } ) ) &
144 valid_bits;
145 in_string = last_bit( in_string_bits, lane_count );
146
147 auto events = one_of<'{', '}', '[', ']', ','>( input ).to_ullong( ) &
148 ~( in_string_bits ^ quote_bits ) & ~escaped_bits &
149 valid_bits;
150 while( events != 0 ) {
151 auto const lane =
152 static_cast<std::size_t>(
153 daw::cxmath::count_trailing_zeros( static_cast<std::uint64_t>( events ) ) );
154 switch( ptr_first[lane] ) {
155 case ',':
156 if( ( primary_depth == 1 ) & ( secondary_depth == 0 ) ) {
157 ++count;
158 }
159 break;
160 case primary_left:
161 ++primary_depth;
162 break;
163 case primary_right:
164 --primary_depth;
165 if( primary_depth == 0 ) {
166 auto const end = ptr_first + lane + 1;
167 if constexpr( not ParseState::is_unchecked_input ) {
168 daw_json_ensure( secondary_depth == 0,
169 ErrorReason::InvalidBracketing,
170 parse_state );
171 }
172 result.last = end;
173 result.counter = count;
174 parse_state.first = end;
175 return result;
176 }
177 break;
178 case secondary_left:
179 ++secondary_depth;
180 break;
181 case secondary_right:
182 --secondary_depth;
183 break;
184 default:
185 DAW_UNREACHABLE( );
186 }
187 events &= events - 1U;
188 }
189 ptr_first += lane_count;
190 }
191
192 if constexpr( not ParseState::is_unchecked_input ) {
193 daw_json_ensure(
194 not in_string, ErrorReason::UnexpectedEndOfData, parse_state );
195 daw_json_ensure( ( primary_depth == 0 ) & ( secondary_depth == 0 ),
196 ErrorReason::InvalidBracketing,
197 parse_state );
198 }
199 result.last = ptr_first;
200 result.counter = count;
201 parse_state.first = ptr_first;
202 return result;
203 }
